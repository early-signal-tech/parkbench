-- ─────────────────────────────────────────────────────────────────────────────
-- OPTION A: Postgres-backed catalog
-- ─────────────────────────────────────────────────────────────────────────────

-- Prerequisites: start Postgres and create the metadata database
-- brew services start postgresql@18
-- psql -d postgres -c "DROP DATABASE IF EXISTS ducklake_v1;"
-- psql -d postgres -c "CREATE DATABASE ducklake_v1;"

-- parkbench setup
-- parkbench run
-- parkbench reset --force

-- Start DuckDB CLI
duckdb

-- Install and load the ducklake extension
INSTALL ducklake;
LOAD ducklake;

-- Attach the Postgres-backed DuckLake catalog
-- DATA_PATH is where Parquet files are written
ATTACH 'ducklake:postgres:dbname=ducklake_v1 host=localhost' AS wh
    (DATA_PATH './ducklake_data', AUTOMATIC_MIGRATION TRUE);

-- ─────────────────────────────────────────────────────────────────────────────
-- OPTION B: DuckDB-backed catalog (no Postgres required)
-- ─────────────────────────────────────────────────────────────────────────────

-- No prerequisites — catalog metadata lives at ./ducklake_data/wh.ducklake

-- parkbench setup --metadata-store duckdb
-- parkbench run   --metadata-store duckdb
-- parkbench reset --metadata-store duckdb --force

-- Start DuckDB CLI
duckdb

-- Install and load the ducklake extension
INSTALL ducklake;
LOAD ducklake;

-- Attach the DuckDB-backed DuckLake catalog
ATTACH 'ducklake:duckdb:./ducklake_data/wh.ducklake' AS wh
    (DATA_PATH './ducklake_data', AUTOMATIC_MIGRATION TRUE);

-- ─────────────────────────────────────────────────────────────────────────────
-- Common queries (both options)
-- ─────────────────────────────────────────────────────────────────────────────

-- Create simple schema table
CREATE TABLE wh.events (
    id          INTEGER,
    ts          TIMESTAMP,
    event_type  VARCHAR
);

-- Create rich schema table
CREATE TABLE wh.events_rich (
    id          INTEGER,
    user_id     VARCHAR,
    event_type  VARCHAR,
    ts          TIMESTAMP,
    payload     JSON,
    metadata    JSON
);

-- Dead-letter table for rejected events (schema drift, etc.)
CREATE TABLE wh.events_rejected (
    rejected_at     TIMESTAMP,
    source_table    VARCHAR,
    anomaly_type    VARCHAR,
    attempted_id    INTEGER,
    error_message   VARCHAR,
    payload         JSON
);

-- Verify tables were created
SHOW TABLES FROM wh;

-- Check DuckLake settings
SELECT * FROM ducklake_settings('wh');

-- Exit when done
.exit
